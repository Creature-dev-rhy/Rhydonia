add_namespace = generic_war


country_event = {
	id = generic_war.1
	title = generic_war.1.t
	desc = generic_war.1.d
	picture = GFX_report_event_soldiers_marching
	
	trigger = {
		is_in_faction = yes
		NOT = { has_war = yes }
		is_subject = no
		
		
		faction_leader = {
			has_war = yes
			
			
			any_enemy_country = {
				OR = {
					is_major = yes
					is_in_faction = yes
				}
			}
		}
		
		
		has_army_manpower = {
			size > 50000
		}
	}
	
	mean_time_to_happen = {
		days = 60
		
		modifier = {
			factor = 0.5
			faction_leader = {
				surrender_progress > 0.2
			}
		}
		
		modifier = {
			factor = 0.7
			has_war_support > 0.5
		}
		
		modifier = {
			factor = 1.5
			has_war_support < 0.3
		}
		
		modifier = {
			factor = 0.6
			has_idea = war_economy
		}
		
		modifier = {
			factor = 0.8
			has_idea = partial_economic_mobilisation
		}
	}
	
	immediate = {
		
		faction_leader = {
			random_enemy_country = {
				limit = {
					OR = {
						is_major = yes
						is_in_faction = yes
						num_of_factories > 20
					}
					NOT = { has_war_with = ROOT }
				}
				save_event_target_as = war_target
			}
		}
	}
	
	
	option = {
		name = generic_war.1.a
		ai_chance = {
			factor = 80
			
			modifier = {
				factor = 1.5
				faction_leader = {
					surrender_progress > 0.2
				}
			}
			
			modifier = {
				factor = 1.3
				has_war_support > 0.5
			}
			
			modifier = {
				factor = 0.5
				has_war_support < 0.2
			}
			
			modifier = {
				factor = 1.2
				has_idea = war_economy
			}
		}
		
		
		create_wargoal = {
			type = take_state_focus
			target = event_target:war_target
			generator = { 5 }
		}
		
		add_war_support = 0.05
		add_political_power = -50
	}
	
	
	option = {
		name = generic_war.1.b
		ai_chance = {
			factor = 20
			
			modifier = {
				factor = 3.0
				has_war_support < 0.2
			}
			
			modifier = {
				factor = 2.0
				has_stability < 0.4
			}
		}
		
		add_stability = 0.03
		
		
		faction_leader = {
			add_opinion_modifier = {
				target = ROOT
				modifier = refused_call_to_arms
			}
		}
	}
}


country_event = {
	id = generic_war.2
	title = generic_war.2.t
	desc = generic_war.2.d
	picture = GFX_report_event_military_planning
	
	trigger = {
		is_in_faction = yes
		NOT = { is_faction_leader = yes }
		has_war = yes
		
		
		any_other_country = {
			is_in_faction_with = ROOT
			NOT = { has_war_with = ROOT }
			NOT = { has_war = yes }
			is_subject = no
		}
		
		
		any_enemy_country = {
			OR = {
				is_major = yes
				is_in_faction = yes
				num_of_factories > 30
			}
		}
	}
	
	mean_time_to_happen = {
		days = 90
		
		modifier = {
			factor = 0.5
			surrender_progress > 0.3
		}
		
		modifier = {
			factor = 0.7
			enemies_strength_ratio > 1.2
		}
		
		modifier = {
			factor = 1.5
			enemies_strength_ratio < 0.7
		}
	}
	
	immediate = {
		
		random_other_country = {
			limit = {
				is_in_faction_with = ROOT
				NOT = { has_war_with = ROOT }
				NOT = { has_war = yes }
				is_subject = no
			}
			save_event_target_as = potential_ally
		}
		
		
		random_enemy_country = {
			limit = {
				OR = {
					is_major = yes
					is_in_faction = yes
					num_of_factories > 30
				}
			}
			save_event_target_as = common_enemy
		}
	}
	
	
	option = {
		name = generic_war.2.a
		ai_chance = {
			factor = 90
			
			modifier = {
				factor = 1.5
				surrender_progress > 0.3
			}
		}
		
		event_target:potential_ally = {
			country_event = { id = generic_war.3 days = 2 }
		}
		
		add_political_power = -25
	}
	
	
	option = {
		name = generic_war.2.b
		ai_chance = {
			factor = 10
		}
		
		add_war_support = 0.03
	}
}
